ALTERNATIVE PYTEST COMMANDS:

  # Run all tests with coverage
  $ pytest -v --cov=.. --cov-report=html

  # Run specific test file
  $ pytest tests/test_payment_flow.py -v

  # Run only fast tests (skip slow tests)
  $ pytest -v -m "not slow"

  # Run specific marker
  $ pytest -v -m "research"    # Research tests
  $ pytest -v -m "payment"     # Payment tests
  $ pytest -v -m "database"    # Database tests

  # Stop on first failure
  $ pytest -x

  # Show test durations
  $ pytest --durations=10
